//! perf study 11 — TEMPORARY deterministic rate counters for the threat
//! channel. Added, measured, and reverted (the diff is kept as
//! `docs/research/perf-artifacts-11/counters11.patch`).
//!
//! Why this exists: the study's headline research claim is a comparison of the
//! FullThreats fold between ARCHON and Stockfish `eca43a97`, read off
//! `perf annotate`'s per-opcode weights. Sampled attribution inside a tight
//! loop is skid-distorted on this box (a `vpdpbusd` reading 0.00% next to a
//! 6.8% `movzwl` in the SAME four-instruction body), so the sampled row count
//! needs a deterministic cross-check before anything is built on it. These
//! counters produce the row count exactly, with no sampling at all.
//!
//! Not behind a cargo feature on purpose: it is meant to be reverted, and a
//! feature flag would invite it to stay.

use std::sync::atomic::{AtomicU64, Ordering};

macro_rules! counters {
    ($($name:ident),+ $(,)?) => {
        $(pub static $name: AtomicU64 = AtomicU64::new(0);)+
        /// Every counter paired with its name, for [`dump`].
        const ALL: &[(&str, &AtomicU64)] = &[$((stringify!($name), &$name)),+];
    };
}

counters! {
    // --- fold shape (the row question) ---
    THR_FOLD_CALLS,    // perspective-folds that reached the kernel
    THR_FOLD_ADD,      // add rows handed to apply_i8_deltas_into
    THR_FOLD_SUB,      // sub rows handed to apply_i8_deltas_into
    THR_COPYONLY,      // perspective-folds that were a pure parent copy
    THR_REFRESH,       // from-scratch refreshes (king-cross + root)
    THR_REFRESH_ROWS,  // active features touched by those refreshes
    // --- what feeds the fold ---
    THR_SPAN_PLIES,    // children whose events were generated
    THR_EVENTS_GEN,    // raw events surviving into the buffer
    THR_PUSH,          // apply_at calls (snapshots taken)
    THR_COMPUTE_CALLS, // at_full -> compute entries that did work
}

#[inline]
pub fn bump(c: &AtomicU64, n: u64) {
    c.fetch_add(n, Ordering::Relaxed);
}

/// Print every counter, raw and per node, to stderr.
pub fn dump(nodes: u64) {
    let n = nodes.max(1) as f64;
    eprintln!("--- perfcount11 (nodes={nodes}) ---");
    for (name, c) in ALL {
        let v = c.load(Ordering::Relaxed);
        eprintln!("{name:<18} {v:>14}  {:>10.4}/node", v as f64 / n);
    }
    let add = THR_FOLD_ADD.load(Ordering::Relaxed);
    let sub = THR_FOLD_SUB.load(Ordering::Relaxed);
    let refresh = THR_REFRESH_ROWS.load(Ordering::Relaxed);
    eprintln!(
        "DERIVED delta_rows/node {:.4}  +refresh_rows/node {:.4}  = total {:.4}",
        (add + sub) as f64 / n,
        refresh as f64 / n,
        (add + sub + refresh) as f64 / n
    );
}
